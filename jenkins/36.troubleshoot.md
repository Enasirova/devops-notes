def helmUpgradeOnCluster(String targetEnvironment, String appName, String ocpProject, String[] products, String targetVersion, String commitId, boolean wantBareMetal = false) {
  def deployParallel = [:]
  def targetEnvironmentForFiles = targetEnvironment.split("-")[0]
  if (targetEnvironmentForFiles.contains('tst')) targetEnvironmentForFiles = 'tst'
  if (targetEnvironmentForFiles.contains('stg')) targetEnvironmentForFiles = 'stg'
  if (targetEnvironmentForFiles.contains('prd')) targetEnvironmentForFiles = 'prd'

  products.each { targetProduct ->
    def serviceName = serviceName(appName, targetProduct)
    def countryCode = countryCodeFromOcpProject(ocpProject)
    def valueFiles = helmValuesFiles(countryCode, targetEnvironmentForFiles, targetProduct, wantBareMetal)
    def deploymentTarget = deploymentTarget(ocpProject, targetProduct)
    deployParallel[deploymentTarget] = {
      echo "Target: ${deploymentTarget} -> ${ocpProject}/${targetEnvironment}"
      sh "oc config use-context ${ocpProject}/${targetEnvironment}"
      //sh "helm version"
      //sh "ls -la ./helm"
      //we set custom annotation to force redeployment if needed (we don't fail the build if the deployment.yaml is not present)
      String pipelineDeployId = commitId.trim()
      sh "sed -i -E \"s/(pipelineDeploymentId: ).+/\\1${pipelineDeployId}/\" ./helm/templates/deployment.yaml || true"
      sh "sed -i -E \"s/(appVersion: ).+/\\1${targetVersion}/\" ./helm/Chart.yaml"
      sh "helm upgrade ${serviceName} helm --set image.tag=${targetVersion} ${valueFiles} --atomic --install"
    }
  }
  return deployParallel
}

/**
 * Prepares list of Helm config files with preserving priority
 * as described here: https://cnfl.myfortuna.eu/display/RND/Helm+Configuration
 */
def helmValuesFiles(String countryCode, String env, String product, boolean wantBareMetal = false) {
  String bareMetalConfigFile = wantBareMetal ? "./helm/values/baremetal-${env}.yaml" : "./helm/values/nonexisting_config_file.yaml"
  String bareMetalConfigFileForCountry = wantBareMetal ? "./helm/values/${countryCode}/baremetal-${env}.yaml" : "./helm/values/nonexisting_config_file.yaml"
  echo "VALUE_FILES ${env} ${countryCode}"

  def files = ""
  [
    "./helm/values/env-${env}.yaml",
    "./helm/values/product-${product}.yaml",
    bareMetalConfigFile,
    "./helm/values/${countryCode}/country-${countryCode}.yaml",
    "./helm/values/${countryCode}/env-${env}.yaml",
    bareMetalConfigFileForCountry,
    "./helm/values/${countryCode}/product-${product}.yaml"
  ].each { filename ->
    if (fileExists(filename)) {
      files = files + " -f ${filename}"
    }
  }
  return files
}

/**
 * Moves image from Quay to OCP
 */
def moveImage(String targetCluster, String appName, String ocpNamespace, String quayOrg, String targetVersion) {
  echo "Moving image into ${ocpNamespace}/${targetCluster}"
  sh "oc config use-context ${ocpNamespace}/${targetCluster}"
  sh "oc tag registry.svc.ifortuna.cz/${quayOrg}/${appName}:${targetVersion} ${ocpNamespace}/${appName}:${targetVersion} --reference-policy='local' --alias=true"
}

def deployToCluster(String targetEnvironment, String appName, String ocpProject, String[] products, Integer rolloutTimeout) {
  def deployParallel = [:]
  products.each { targetProduct ->
    def serviceName = serviceName(appName, targetProduct)
    def deploymentTarget = deploymentTarget(ocpProject, targetProduct)
    deployParallel[deploymentTarget] = {
      echo "Deploying ${deploymentTarget}"
      sh "oc config use-context ${ocpProject}/${targetEnvironment}"
      sh "oc rollout restart deployment/dc-${serviceName}"
      timeout(time: rolloutTimeout, unit: 'SECONDS') {
        sh "oc rollout status deployment/dc-${serviceName}"
      }
    }
  }
  return deployParallel
}

/**
 * Generates service name from app name and product (if available)
 */
def serviceName(String appName, String product) {
  return product == '' ? "${appName}" : "${appName}-${product}"
}

/**
 * Helper for getting deployment target from project name and product (if available)
 */
def deploymentTarget(String project, String product) {
  return product == '' ? project : "${project}-${product}"
}

/**
 * Gets country code from the standard OCP project name
 */
def countryCodeFromOcpProject(String ocpProject) {
  return ocpProject.split("-")[1]
}

/**
 * Gets namespace from the standard OCP project name
 */
def namespaceFromOcpProject(String ocpProject) {
  return ocpProject.split("-")[0]
}

boolean isSnapshotVersion() {
  return readMavenPom().getVersion().contains('-SNAPSHOT')
}

boolean isReleaseCandidateVersion() {
  return readMavenPom().getVersion().contains('-RC')
}

boolean isUnstableVersion() {
  return isSnapshotVersion()
}

boolean isStableVersion() {
  return !isUnstableVersion()
}

String randomManualCommitId() {
  return "MANUAL-${new Date().format('HHmmss')}"
}
